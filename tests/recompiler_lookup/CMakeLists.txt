cmake_minimum_required(VERSION 3.20)
project(recompiler_lookup LANGUAGES C CXX)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION FALSE)
foreach(tag baseline candidate)
  add_executable(lookup_gen_${tag} generate.cpp)
  target_compile_features(lookup_gen_${tag} PRIVATE cxx_std_20)
  if(tag STREQUAL "baseline")
    target_include_directories(lookup_gen_${tag} PRIVATE "${LOOKUP_BASELINE_INCLUDE}")
  else()
    target_include_directories(lookup_gen_${tag} PRIVATE "${LOOKUP_CANDIDATE_INCLUDE}")
    target_compile_definitions(lookup_gen_${tag} PRIVATE LOOKUP_CANDIDATE=1)
  endif()
  set(files)
  foreach(range low high)
    foreach(allocation on fail)
      list(APPEND files "${CMAKE_CURRENT_BINARY_DIR}/gen/${tag}/${tag}_${range}_${allocation}.c")
    endforeach()
  endforeach()
  add_custom_command(OUTPUT ${files} COMMAND lookup_gen_${tag} "${CMAKE_CURRENT_BINARY_DIR}/gen/${tag}"
                     DEPENDS lookup_gen_${tag} VERBATIM)
  list(APPEND all_generated ${files})
endforeach()
add_executable(lookup_test runner.cpp ${all_generated})
target_compile_features(lookup_test PRIVATE cxx_std_20 c_std_11)
foreach(target lookup_gen_baseline lookup_gen_candidate lookup_test)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /Ob2 /bigobj /fp:precise)
  else()
    target_compile_options(${target} PRIVATE -O2)
  endif()
endforeach()
